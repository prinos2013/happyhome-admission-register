3-Minute Setup Instructions:
  1. Create a new spreadsheet on Google Drive.
  2. Go to Extensions > Apps Script.
  3. Replace the script with the code below, then click Deploy > New deployment.
  4. Select Web app, execute as Me, and set "Who has access" to Anyone.
  5. Copy the web app URL and paste it into the field above!
Google Apps Script (`doPost` Code): 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Auto-create headers if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "S/N", "Academic Year", "Term", "Name of Student", "Class",
        "Sex", "Date of Birth", "Age", "Admission No.", "NHIS NO.",
        "Physical Deformity", "Type of Deformity", "Hometown",
        "Name of Real father", "Is father alive?", "Name of Real Mother", "Is mother alive?",
        "Name of Guardian", "Community", "Contact 1", "Contact 2",
        "Date of Leaving", "Reason for Leaving", "Timestamp"
      ]);
    }

    var list = Array.isArray(data) ? data : [data];
    list.forEach(function(r) {
      sheet.appendRow([
        r.sn || "",
        r.academicYear || "",
        r.term || "",
        r.name || "",
        r.studentClass || "",
        r.sex || "",
        r.dob || "",
        r.age || "",
        r.admissionNo || "",
        r.nhisNo || "",
        r.physicalDeformity || "",
        r.typeOfDeformity || "",
        r.hometown || "",
        r.fatherName || "",
        r.fatherAlive || "",
        r.motherName || "",
        r.motherAlive || "",
        r.guardianName || "",
        r.community || "",
        r.contact1 || "",
        r.contact2 || "",
        r.dateOfLeaving || "",
        r.reasonForLeaving || "",
        new Date()
      ]);
    });

    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

🐙 How to Host on GitHub Pages (Free & Fast)

This web application is completely self-contained in a single file (index.html). You can host it on GitHub Pages in under 2 minutes:

  1. Log in to your GitHub account and click New Repository.
  2. Name the repository (e.g. prinesther-student-register) and set it to Public.
  3. Click Upload an existing file and upload this file named as index.html.
  4. Commit changes.
  5. Go to repository Settings > Pages.
  6. Under Branch, select main (or master) and folder / (root), then click Save.
  7. Within seconds, your live address will be:
    https://<your-username>.github.io/prinesther-student-register/
